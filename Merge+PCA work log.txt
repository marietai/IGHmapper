#Get list of variant positions from GoalPost
plink2 --bfile 1000G_clean --write-snplist --out 1000G_vars
plink2 --bfile GoalPost_nodup --write-snplist --out goalpost_vars

# Compare SNPs and save overlaps
# Load SNP lists
with open("GoalPost_vars.snplist", "r") as f:
    snps_goalpost = set(line.strip() for line in f if line.strip())

with open("1000G_vars.snplist", "r") as f:
    snps_1000G = set(line.strip() for line in f if line.strip())

# Find intersection
shared_snps = snps_goalpost.intersection(snps_1000G)

# Save to shared.snplist
with open("shared.snplist", "w") as out:
    for snp in sorted(shared_snps):
        out.write(snp + "\n")

print(f"Shared SNPs: {len(shared_snps)} written to shared.snplist")

plink2 --bfile GoalPost_nodup --extract shared.snplist --make-bed --out GoalPost_overlap
plink2 --bfile 1000G_clean --extract shared.snplist --make-bed --out 1000G_overlap

# Merge
echo 1000G_overlap.bed 1000G_overlap.bim 1000G_overlap.fam > mergelist.txt
plink1.9 --bfile GoalPost_overlap --merge-list mergelist.txt --make-bed --out merged_overlap

# QC 
plink2 --bfile merged_overlap ^
       --geno 0.05 --maf 0.05 --mind 0.05 ^
       --make-bed --out merged_QC

# LD Pruning
plink2 --bfile merged_QC ^
       --indep-pairwise 50 5 0.2 ^
       --out pruned

plink2 --bfile merged_QC ^
       --extract pruned.prune.in ^
       --make-bed --out merged_pruned

# PCA
plink2 --bfile merged_pruned --pca 10 --out pca_results

#Plot in R
library(GGally)
library(tidyverse)
library(ggplot2)

# Load PCA
pca <- read.table("pca_results.eigenvec", header = FALSE)
colnames(pca) <- c("FID", "IID", paste0("PC", 1:10))

# Load .fam to assign origin labels
fam <- read.table("merged_overlap.fam", header = FALSE)
colnames(fam) <- c("FID", "IID", "PID", "MID", "SEX", "PHENO")
fam$Origin <- ifelse(fam$FID == "GoalPost", "GoalPost", "1000G")

# Merge PCA with origin info
pca <- merge(pca, fam[, c("FID", "IID", "Origin")], by = c("FID", "IID"))
table(pca$Origin)

# Ensure Origin is a factor for color mapping
pca$Origin <- factor(pca$Origin, levels = c("GoalPost", "1000G"))

# Plot
p <- ggpairs(
  pca,
  columns = 3:7,  # PC1 to PC5
  mapping = aes(color = Origin),
  upper = list(continuous = wrap("points", alpha = 0.6, size = 0.8)),
  lower = list(continuous = wrap("points", alpha = 0.6, size = 0.8)),
  diag = list(continuous = wrap("densityDiag", alpha = 0.4))
)

# Force legend to appear in all subplots
for(i in 1:p$ncol) {
  for(j in 1:p$nrow) {
    p[i, j] <- p[i, j] + theme(legend.position = "right")
  }
}

# Show the plot
print(p)

# Save plot

       plot = p, width = 12, height = 12, dpi = 300)

#  Identify furthest GoalPost sample

# Re-load PCA file
pca <- read.table("pca_results.eigenvec", header = FALSE)
colnames(pca) <- c("FID", "IID", paste0("PC", 1:10))

# Re-load fam file to assign origin
fam <- read.table("merged_QC.fam", header = FALSE)
colnames(fam) <- c("FID", "IID", "PID", "MID", "SEX", "PHENO")
fam$Origin <- ifelse(fam$FID == "GoalPost", "GoalPost", "1000G")

# Merge to get origin
pca <- merge(pca, fam[, c("FID", "IID", "Origin")], by = c("FID", "IID"))

# Calculate Euclidean distance from the GoalPost centroid
goalpost <- subset(pca, Origin == "GoalPost")
goalpost$distance <- sqrt(rowSums(goalpost[, paste0("PC", 1:6)]^2))

# Identify furthest GoalPost sample
outlier <- goalpost[which.max(goalpost$distance), c("FID", "IID", paste0("PC", 1:6), "distance")]
print(outlier)
