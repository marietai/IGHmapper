# Convert VCF to PLINK files
for chr in {1..22} X Y; do
    plink2 --vcf ALL.chr${chr}.phase3_shapeit2_mvncall_integrated_v5a.20130502.genotypes.vcf.gz \
           --make-bed \
           --out 1000G_chr${chr}_plink
done


#Adds a prefix based on chromosome (Chromosome_SNPID).
for chr in {1..22} X Y; do
    awk '{print $1, $1"_"$2, $3, $4, $5, $6}' 1000G_chr${chr}_plink.bim > 1000G_chr${chr}_SNPid.bim
    mv 1000G_chr${chr}_plink.bim 1000G_chr${chr}_plink.bim.bak  # Backup original
    mv 1000G_chr${chr}_SNPid.bim 1000G_chr${chr}_plink.bim      # Replace with renamed
done



# remove duplicate SNPs across all chromosomes.
for chr in {1..22} X Y; do
    cut -f2 1000G_chr${chr}_plink.bim | sort | uniq -d > 1000G_chr${chr}_duplicates.txt
    plink --bfile 1000G_chr${chr}_plink --exclude 1000G_chr${chr}_duplicates.txt --make-bed --out 1000G_chr${chr}_nodups
done
