# ==========================================================
# Step 3: Check for Sex Discrepancies
# ==========================================================
## Step 3.1: Compute Hardy-Weinberg Equilibrium (HWE) on X Chromosome
plink2 --bfile /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step2_mind \\
       --chr X \\
       --hardy \\
       --out /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step3_xhardy

## Step 3.2: Compute Per-Individual Heterozygosity on X Chromosome
plink2 --bfile /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step2_mind \\
       --chr X \\
       --freq counts \\
       --out /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step3_xcounts

## Step 3.3: Extract Per-Individual Heterozygosity from Allele Counts
awk '{if (NR > 1 && $5 > 0) print $1, $2, $6/$5}' /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step3_xcounts.acount > /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step3_xhet.txt

## Step 3.4: Classify Individuals by X Chromosome Heterozygosity
awk '{if ($3 < 0.2) print $1, $2, "Likely Male"; else if ($3 > 0.8) print $1, $2, "Likely Female"; else print $1, $2, "Ambiguous"}' /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step3_xhet.txt > /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/sex_predictions.txt

## Step 3.5: Compare Computed Sex with Recorded Sex
sort /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/sex_predictions.txt > sorted_sex_predictions.txt
sort /rds/general/user/wt1224/home/IG-heavy-chain-/my_genotype_data/GoalPost_PhenoSex.txt > sorted_GoalPost_PhenoSex.txt
join -1 1 -2 1 sorted_sex_predictions.txt sorted_GoalPost_PhenoSex.txt | awk '\$3 != \$4'

## Step 3.6: Remove Individuals with Ambiguous Sex
awk '\$3 == "Ambiguous"' /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/sex_predictions.txt > /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/fail_sexcheck.txt

plink2 --bfile /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step2_mind \\
       --remove /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/fail_sexcheck.txt \\
       --make-bed \\
       --out /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step4_sex_filtered

## Step 4.2: Compute Autosomal Heterozygosity
plink2 --bfile /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step4_sex_filtered \
       --autosome \
       --het \
       --out /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step4_het

## Step 4.3: Compute Outlier Thresholds (Mean ± 3 SD)
awk '{if (NR > 1) print }' /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step4_het.het | \
    python3 -c "
import sys
import numpy as np
data = [float(line.strip()) for line in sys.stdin]
mean = np.mean(data)
std = np.std(data)
lower_bound = mean - 3 * std
upper_bound = mean + 3 * std
print(f'Heterozygosity Mean: {mean}, Std: {std}')
print(f'Outlier Range: <{lower_bound} or >{upper_bound}')
"

## Step 4.4: Identify and Flag Outliers
awk -v lb=-0.04774 -v ub=0.03381 '{if (NR > 1 && ( < lb ||  > ub)) print , }' /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step4_het.het > /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/fail_hetcheck.txt

## Step 4.5: Remove Heterozygosity Outliers
plink2 --bfile /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step4_sex_filtered \

       --make-bed \
       --out /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step5_het_filtered

## Step 6.3: Identify Population Outliers (Beyond ±6SD)
awk '{if (NR > 1) print $3, $4}' /rds/general/user/wt1224/home/IG-heavy-chain-/16Mar_QC_SC_PCA/qc_step6_pca.eigenvec | \\
    python3 -c "
import sys
import numpy as np
data = np.array([list(map(float, line.split())) for line in sys.stdin])
mean_pca1, mean_pca2 = np.mean(data, axis=0)
std_pca1, std_pca2 = np.std(data, axis=0)
outliers = [(i+1, x[0], x[1]) for i, x in enumerate(data) if abs(x[0] - mean_pca1) > 6 * std_pca1 or abs(x[1] - mean_pca2) > 6 * std_pca2]
for outlier in outliers:
    print(f'Outlier {outlier[0]}: PC1={outlier[1]}, PC2={outlier[2]}')
"
