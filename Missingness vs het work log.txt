## Plot with R

library(ggplot2)
library(GGally)
library(dplyr)
library(readr)
library(ggrepel)

# Read files
smiss <- read_table("C:/Users/marietai/Desktop/GoalPost/GP/GoalPost_LDpruned_noHighLD.smiss") %>%
  rename(FID = `#FID`)

het <- read_table("C:/Users/marietai/Desktop/GoalPost/GP/GoalPost_LDpruned_noHighLD.het") %>%
  rename(FID = `#FID`)

# Merge and compute values
merged <- inner_join(smiss, het, by = c("FID", "IID")) %>%
  mutate(
    log10_missing = log10(F_MISS),
    F_inbreeding = F
  )

plot_data <- merged %>%
  select(log10_missing, F_inbreeding)

# Custom lower panel with h=0 line
custom_lower <- function(data, mapping, ...) {
  ggplot(data = data, mapping = mapping) +
    geom_point(alpha = 0.7) +
    geom_smooth(method = "lm", se = FALSE, size = 0.7, color = "blue", ...) +
    geom_hline(yintercept = 0, linetype = "dashed", color = "red") +
    theme_minimal()
}

# Identify outliers: ±3 SD
mean_log10miss <- mean(merged$log10_missing, na.rm = TRUE)
sd_log10miss <- sd(merged$log10_missing, na.rm = TRUE)

mean_F <- mean(merged$F_inbreeding, na.rm = TRUE)
sd_F <- sd(merged$F_inbreeding, na.rm = TRUE)

merged <- merged %>%
  mutate(
    is_outlier = abs(log10_missing - mean_log10miss) > 3 * sd_log10miss |
                 abs(F_inbreeding - mean_F) > 3 * sd_F
  )

# Update the plot with PC1–PC6
p <- ggpairs(
  pca,
  columns = 3:8,  # PC1 to PC6
  mapping = aes(color = Origin),
  upper = list(continuous = wrap("points", alpha = 0.6, size = 0.8)),
  lower = list(continuous = wrap("points", alpha = 0.6, size = 0.8)),
  diag = list(continuous = wrap("densityDiag", alpha = 0.4))
)

# Force legend on all subplots
for(i in 1:p$ncol) {
  for(j in 1:p$nrow) {
    p[i, j] <- p[i, j] + theme(legend.position = "right")
  }
}

# View or save
print(p)
